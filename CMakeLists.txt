cmake_minimum_required(VERSION 3.16)
project(regex_shortener VERSION 6.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/catalog")
configure_file(catalog/seed.regex "${CMAKE_CURRENT_BINARY_DIR}/catalog/seed.regex" COPYONLY)
add_executable(regex-shortener src/main.cpp)
target_compile_features(regex-shortener PRIVATE cxx_std_17)
target_link_libraries(regex-shortener PRIVATE Threads::Threads)
target_compile_options(regex-shortener PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(regex-core-tests tests/core.cpp)
  target_compile_features(regex-core-tests PRIVATE cxx_std_17)
  target_compile_options(regex-core-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  target_link_libraries(regex-core-tests PRIVATE Threads::Threads)
  add_test(NAME mathematical-core COMMAND regex-core-tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test.py $<TARGET_FILE:regex-shortener>)
    add_test(NAME optimality-oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/optimality_oracle.py $<TARGET_FILE:regex-shortener>)
  endif()
endif()

if(BUILD_TESTING)
  foreach(extra finite automata_search generalization algebra_search semantic_synthesis closure_search profile_search unary_sets unary_evolution optimality context_loops)
    add_executable(regex-${extra}-tests tests/${extra}.cpp)
    target_compile_features(regex-${extra}-tests PRIVATE cxx_std_17)
    target_compile_options(regex-${extra}-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    target_link_libraries(regex-${extra}-tests PRIVATE Threads::Threads)
    add_test(NAME ${extra} COMMAND regex-${extra}-tests)
  endforeach()
endif()
